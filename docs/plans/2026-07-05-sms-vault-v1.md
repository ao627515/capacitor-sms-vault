```kotlin
package com.oaalab.capacitorsmsvault

import android.content.Context
import android.database.Cursor
import android.provider.Telephony
import com.getcapacitor.JSArray
import com.getcapacitor.JSObject
import com.getcapacitor.Logger

class SmsReader(private val context: Context) {

    companion object {
        private const val TAG = "SmsReader"
        private const val DEFAULT_LIMIT = 100
    }

    /**
     * Lit les SMS de l'inbox avec filtres optionnels.
     * @param address Filtrer par numéro d'expéditeur (LIKE)
     * @param since Timestamp ms minimum (date >=)
     * @param limit Nombre max de résultats
     */
    fun getSmsList(address: String?, since: Long?, limit: Int?): JSArray {
        val result = JSArray()
        val effectiveLimit = limit ?: DEFAULT_LIMIT

        val selection = buildSelection(address, since)
        val selectionArgs = buildSelectionArgs(address, since)

        val projection = arrayOf(
            Telephony.Sms._ID,
            Telephony.Sms.ADDRESS,
            Telephony.Sms.BODY,
            Telephony.Sms.DATE,
            Telephony.Sms.TYPE
        )

        var cursor: Cursor? = null
        try {
            cursor = context.contentResolver.query(
                Telephony.Sms.CONTENT_URI,
                projection,
                selection,
                selectionArgs,
                "${Telephony.Sms.DATE} DESC LIMIT $effectiveLimit"
            )

            cursor?.let {
                val idIndex = it.getColumnIndexOrThrow(Telephony.Sms._ID)
                val addressIndex = it.getColumnIndexOrThrow(Telephony.Sms.ADDRESS)
                val bodyIndex = it.getColumnIndexOrThrow(Telephony.Sms.BODY)
                val dateIndex = it.getColumnIndexOrThrow(Telephony.Sms.DATE)
                val typeIndex = it.getColumnIndexOrThrow(Telephony.Sms.TYPE)

                while (it.moveToNext()) {
                    val smsObj = JSObject().apply {
                        put("id", it.getString(idIndex) ?: "")
                        put("address", it.getString(addressIndex) ?: "")
                        put("body", it.getString(bodyIndex) ?: "")
                        put("date", it.getLong(dateIndex))
                        put("type", mapType(it.getInt(typeIndex)))
                    }
                    result.put(smsObj)
                }
            }
        } catch (e: Exception) {
            Logger.error(TAG, "Failed to read SMS list", e)
        } finally {
            cursor?.close()
        }

        return result
    }

    private fun buildSelection(address: String?, since: Long?): String? {
        val conditions = mutableListOf<String>()
        if (address != null) conditions.add("${Telephony.Sms.ADDRESS} LIKE ?")
        if (since != null) conditions.add("${Telephony.Sms.DATE} >= ?")
        return if (conditions.isEmpty()) null else conditions.joinToString(" AND ")
    }

    private fun buildSelectionArgs(address: String?, since: Long?): Array<String>? {
        val args = mutableListOf<String>()
        if (address != null) args.add("%$address%")
        if (since != null) args.add(since.toString())
        return if (args.isEmpty()) null else args.toTypedArray()
    }

    private fun mapType(type: Int): String {
        return when (type) {
            Telephony.Sms.MESSAGE_TYPE_SENT -> "sent"
            else -> "inbox"
        }
    }
}
```

package com.oaalab.capacitorsmsvault

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.provider.Telephony
import com.getcapacitor.JSObject
import com.getcapacitor.Logger
import java.lang.ref.WeakReference

class SmsReceiver : BroadcastReceiver() {

    companion object {
        private const val TAG = "SmsReceiver"
    }

    // WeakReference pour éviter les fuites mémoire
    var pluginRef: WeakReference<SmsVaultPlugin>? = null

    override fun onReceive(context: Context?, intent: Intent?) {
        if (intent?.action != Telephony.Sms.Intents.SMS_RECEIVED_ACTION) return

        val plugin = pluginRef?.get() ?: run {
            Logger.warn(TAG, "Plugin reference lost, ignoring SMS")
            return
        }

        val messages = Telephony.Sms.Intents.getMessagesFromIntent(intent)
        if (messages.isNullOrEmpty()) return

        for (smsMessage in messages) {
            try {
                val smsObj = JSObject().apply {
                    put("id", "")  // pas disponible via PDU
                    put("address", smsMessage.displayOriginatingAddress ?: "")
                    put("body", smsMessage.messageBody ?: "")
                    put("date", smsMessage.timestampMillis)
                    put("type", "inbox")
                }
                plugin.notifyListeners("smsReceived", smsObj)
                Logger.info(TAG, "SMS received from: ${smsMessage.displayOriginatingAddress}")
            } catch (e: Exception) {
                Logger.error(TAG, "Failed to parse SMS PDU", e)
            }
        }
    }
}

**Step 1: Réécrire SmsVaultPlugin.kt**

```kotlin
package com.oaalab.capacitorsmsvault

import android.Manifest
import android.app.Activity
import android.content.Intent
import android.content.IntentFilter
import android.os.Build
import android.provider.Telephony
import com.getcapacitor.JSObject
import com.getcapacitor.Logger
import com.getcapacitor.PermissionState
import com.getcapacitor.Plugin
import com.getcapacitor.PluginCall
import com.getcapacitor.PluginMethod
import com.getcapacitor.annotation.ActivityCallback
import com.getcapacitor.annotation.CapacitorPlugin
import com.getcapacitor.annotation.Permission
import com.getcapacitor.annotation.PermissionCallback
import java.lang.ref.WeakReference

@CapacitorPlugin(
    name = "SmsVault",
    permissions = [
        Permission(
            alias = "readSms",
            strings = [Manifest.permission.READ_SMS]
        ),
        Permission(
            alias = "receiveSms",
            strings = [Manifest.permission.RECEIVE_SMS]
        )
    ]
)
class SmsVaultPlugin : Plugin() {

    companion object {
        private const val TAG = "SmsVaultPlugin"
        private const val REQUEST_DEFAULT_SMS_APP = 1001
    }

    private val smsReceiver = SmsReceiver()
    private var isListening = false
    private var smsReader: SmsReader? = null
    private var pendingDefaultSmsCall: PluginCall? = null

    override fun load() {
        smsReader = SmsReader(context)
        smsReceiver.pluginRef = WeakReference(this)
    }

    // ─── F1 : Permissions ───────────────────────────────────────────────────

    @PluginMethod
    fun checkPermissions(call: PluginCall) {
        val readState = getPermissionState("readSms")
        val receiveState = getPermissionState("receiveSms")
        call.resolve(buildPermissionResult(readState, receiveState))
    }

    @PluginMethod
    fun requestPermissions(call: PluginCall) {
        if (getPermissionState("readSms") == PermissionState.GRANTED &&
            getPermissionState("receiveSms") == PermissionState.GRANTED) {
            checkPermissions(call)
            return
        }
        requestPermissionForAliases(arrayOf("readSms", "receiveSms"), call, "permissionsCallback")
    }

    @PermissionCallback
    private fun permissionsCallback(call: PluginCall) {
        val readState = getPermissionState("readSms")
        val receiveState = getPermissionState("receiveSms")
        call.resolve(buildPermissionResult(readState, receiveState))
    }

    private fun buildPermissionResult(
        readState: PermissionState?,
        receiveState: PermissionState?
    ): JSObject {
        return JSObject().apply {
            put("readSms", readState?.toString() ?: "prompt")
            put("receiveSms", receiveState?.toString() ?: "prompt")
        }
    }

    // ─── F2 : Default SMS App ───────────────────────────────────────────────

    @PluginMethod
    fun isDefaultSmsApp(call: PluginCall) {
        val defaultPackage = Telephony.Sms.getDefaultSmsPackage(context)
        val isDefault = defaultPackage == context.packageName
        call.resolve(JSObject().put("isDefault", isDefault))
    }

    @PluginMethod
    fun requestDefaultSmsApp(call: PluginCall) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            // Android 10+ : utiliser RoleManager
            val roleManager = context.getSystemService(android.app.role.RoleManager::class.java)
            if (roleManager.isRoleHeld(android.app.role.RoleManager.ROLE_SMS)) {
                call.resolve(JSObject().put("isDefault", true))
                return
            }
            val intent = roleManager.createRequestRoleIntent(android.app.role.RoleManager.ROLE_SMS)
            pendingDefaultSmsCall = call
            startActivityForResult(call, intent, "defaultSmsCallback")
        } else {
            val intent = Intent(Telephony.Sms.Intents.ACTION_CHANGE_DEFAULT_SMS_APP)
            intent.putExtra(Telephony.Sms.Intents.EXTRA_PACKAGE_NAME, context.packageName)
            pendingDefaultSmsCall = call
            startActivityForResult(call, intent, "defaultSmsCallback")
        }
    }

    @ActivityCallback
    private fun defaultSmsCallback(call: PluginCall?, result: androidx.activity.result.ActivityResult) {
        val resolveCall = call ?: pendingDefaultSmsCall ?: return
        pendingDefaultSmsCall = null
        val isDefault = Telephony.Sms.getDefaultSmsPackage(context) == context.packageName
        resolveCall.resolve(JSObject().put("isDefault", isDefault))
    }

    // ─── F3 : Lecture historique ────────────────────────────────────────────

    @PluginMethod
    fun getSmsList(call: PluginCall) {
        if (getPermissionState("readSms") != PermissionState.GRANTED) {
            call.reject("PERMISSION_DENIED", "READ_SMS permission is required")
            return
        }
        try {
            val address = call.getString("address")
            val since = if (call.hasOption("since")) call.getLong("since") else null
            val limit = if (call.hasOption("limit")) call.getInt("limit") else null

            val messages = smsReader!!.getSmsList(address, since, limit?.toLong())
            call.resolve(JSObject().put("messages", messages))
        } catch (e: Exception) {
            Logger.error(TAG, "getSmsList failed", e)
            call.reject("READ_FAILED", e.message)
        }
    }

    // ─── F4 : Interception temps réel ───────────────────────────────────────

    @PluginMethod
    fun startListening(call: PluginCall) {
        if (getPermissionState("receiveSms") != PermissionState.GRANTED) {
            call.reject("PERMISSION_DENIED", "RECEIVE_SMS permission is required")
            return
        }
        if (isListening) {
            call.resolve()
            return
        }
        try {
            val filter = IntentFilter(Telephony.Sms.Intents.SMS_RECEIVED_ACTION).apply {
                priority = 999
            }
            context.registerReceiver(smsReceiver, filter)
            isListening = true
            Logger.info(TAG, "SMS listening started")
            call.resolve()
        } catch (e: Exception) {
            Logger.error(TAG, "startListening failed", e)
            call.reject("START_FAILED", e.message)
        }
    }

    @PluginMethod
    fun stopListening(call: PluginCall) {
        if (!isListening) {
            call.resolve()
            return
        }
        try {
            context.unregisterReceiver(smsReceiver)
            isListening = false
            Logger.info(TAG, "SMS listening stopped")
        } catch (e: Exception) {
            Logger.warn(TAG, "stopListening: receiver was not registered")
        }
        call.resolve()
    }

    override fun handleOnDestroy() {
        if (isListening) {
            try {
                context.unregisterReceiver(smsReceiver)
                isListening = false
            } catch (e: Exception) {
                Logger.warn(TAG, "Cleanup: receiver was not registered")
            }
        }
        super.handleOnDestroy()
    }
}
```

## Configuration Android

### Permissions dans le Manifest de l'app hôte

Ajouter dans `android/app/src/main/AndroidManifest.xml` de **votre application** :

<uses-permission android:name="android.permission.READ_SMS" />

### App SMS par défaut (requis pour READ_SMS sur Android 10+)

Sur Android 10+, votre app doit être l'application SMS par défaut pour lire l'historique SMS. Utilisez `requestDefaultSmsApp()` pour déclencher le dialogue système.

## Exemples d'utilisation

### Vérifier et demander les permissions

```typescript
import { SmsVault } from 'capacitor-sms-vault';

const status = await SmsVault.checkPermissions();
if (status.readSms !== 'granted') {
  await SmsVault.requestPermissions();
}
```

### Lire l'historique SMS

```typescript
const { messages } = await SmsVault.getSmsList({
  limit: 20,
  since: Date.now() - 7 * 24 * 60 * 60 * 1000, // 7 derniers jours
});
console.log(messages);
```

### Intercepter les SMS en temps réel

```typescript
await SmsVault.addListener('smsReceived', (sms) => {
  console.log('SMS reçu de', sms.address, ':', sms.body);
});
await SmsVault.startListening();

// Plus tard :
await SmsVault.stopListening();
await SmsVault.removeAllListeners();
```

## Conformité Google Play

> ⚠️ **Important** : Les permissions `READ_SMS` et `RECEIVE_SMS` nécessitent une déclaration dans la Play Console (formulaire "Permissions Declaration"). L'application hôte doit justifier l'usage de ces permissions et être l'app SMS par défaut pour y accéder.

import { SmsVault } from 'capacitor-sms-vault';

// ─── Permissions ────────────────────────────────────────────────────────────

document.getElementById('btnCheckPerms').addEventListener('click', async () => {
  const status = await SmsVault.checkPermissions();
  document.getElementById('permStatus').textContent =
    `readSms: ${status.readSms} | receiveSms: ${status.receiveSms}`;
});

document.getElementById('btnRequestPerms').addEventListener('click', async () => {
  const status = await SmsVault.requestPermissions();
  document.getElementById('permStatus').textContent =
    `readSms: ${status.readSms} | receiveSms: ${status.receiveSms}`;
});

document.getElementById('btnCheckDefault').addEventListener('click', async () => {
  const { isDefault } = await SmsVault.isDefaultSmsApp();
  document.getElementById('permStatus').textContent =
    `App SMS par défaut : ${isDefault ? '✅ Oui' : '❌ Non'}`;
});

document.getElementById('btnRequestDefault').addEventListener('click', async () => {
  const { isDefault } = await SmsVault.requestDefaultSmsApp();
  document.getElementById('permStatus').textContent =
    `App SMS par défaut : ${isDefault ? '✅ Oui' : '❌ Non'}`;
});

// ─── Historique SMS ─────────────────────────────────────────────────────────

document.getElementById('btnReadSms').addEventListener('click', async () => {
  const list = document.getElementById('smsList');
  list.innerHTML = '<ion-item><ion-label>Chargement...</ion-label></ion-item>';
  try {
    const { messages } = await SmsVault.getSmsList({ limit: 10 });
    list.innerHTML = messages.length === 0
      ? '<ion-item><ion-label>Aucun SMS trouvé</ion-label></ion-item>'
      : messages.map(sms => `
          <ion-item>
            <ion-label>
              <h3>${sms.address}</h3>
              <p>${sms.body.substring(0, 80)}${sms.body.length > 80 ? '...' : ''}</p>
              <p><small>${new Date(sms.date).toLocaleString()}</small></p>
            </ion-label>
          </ion-item>
        `).join('');
  } catch (e) {
    list.innerHTML = `<ion-item><ion-label color="danger">Erreur : ${e.message}</ion-label></ion-item>`;
  }
});

// ─── Écoute temps réel ──────────────────────────────────────────────────────

let liveListener = null;

document.getElementById('btnStartListen').addEventListener('click', async () => {
  await SmsVault.startListening();
  liveListener = await SmsVault.addListener('smsReceived', (sms) => {
    const list = document.getElementById('liveSmsList');
    const item = document.createElement('ion-item');
    item.innerHTML = `
      <ion-label>
        <h3>${sms.address}</h3>
        <p>${sms.body}</p>
        <p><small>${new Date(sms.date).toLocaleString()}</small></p>
      </ion-label>
    `;
    list.prepend(item);
  });
});

document.getElementById('btnStopListen').addEventListener('click', async () => {
  await SmsVault.stopListening();
  if (liveListener) {
    await liveListener.remove();
    liveListener = null;
  }
});
```
